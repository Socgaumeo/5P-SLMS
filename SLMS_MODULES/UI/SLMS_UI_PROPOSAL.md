### 1.2 Danh sách Modules

| Module | Chức năng | Tables/Views chính |
|--------|-----------|-------------------|
| **M1: Foundation** | Users, Roles, Master Data | users, customers, vendors, drivers, vehicles, routes |
| **M2: Job Management** | Tạo/quản lý Jobs qua Chat UI | jobs, job_services, job_events |
| **M3: Service Management** | Quản lý dịch vụ theo loại | v_trucking, v_warehouse, v_customs, v_packing |
| **M4: Financial** | Giá, Statements, Reports | vendor_rates, customer_rates, statements |
| **M5: AI Integration** | Chat UI + AI Processing | chat_sessions, chat_messages, message_templates |

### 4.2 Dynamic Service Menu Hook

```typescript
// hooks/useServiceVisibility.ts

import { useQuery } from '@tanstack/react-query';
import { api } from '@/lib/api';

interface ServiceCounts {
  trucking: number;
  warehouse: number;
  customs: number;
  packing: number;
}

export function useServiceVisibility() {
  const { data, isLoading } = useQuery<ServiceCounts>({
    queryKey: ['serviceCounts'],
    queryFn: async () => {
      const response = await api.get('/api/services/counts');
      return response.data;
    },
    refetchInterval: 60000, // Refresh every minute
  });

  return {
    isLoading,
    showTrucking: (data?.trucking ?? 0) > 0,
    showWarehouse: (data?.warehouse ?? 0) > 0,
    showCustoms: (data?.customs ?? 0) > 0,
    showPacking: (data?.packing ?? 0) > 0,
  };
}

// Usage in Sidebar:
function Sidebar() {
  const { showTrucking, showWarehouse, showCustoms, showPacking } = useServiceVisibility();

  return (
    <nav>
      {/* Always show */}
      <NavItem icon="📊" label="Dashboard" href="/dashboard" />
      <NavItem icon="📋" label="Jobs" href="/jobs" />
      
      {/* Conditional service menus */}
      {showTrucking && <NavItem icon="🚚" label="Trucking" href="/trucking" />}
      {showWarehouse && <NavItem icon="🏭" label="Warehouse" href="/warehouse" />}
      {showCustoms && <NavItem icon="📋" label="Customs" href="/customs" />}
      {showPacking && <NavItem icon="📦" label="Packing" href="/packing" />}
    </nav>
  );
}
```

```python
# GET /api/services/counts
@router.get("/services/counts")
async def get_service_counts(db: Session = Depends(get_db)):
    """Return count of each service type for dynamic menu visibility"""
    
    return {
        "trucking": db.execute(
            "SELECT COUNT(*) FROM v_trucking_services"
        ).scalar(),
        "warehouse": db.execute(
            "SELECT COUNT(*) FROM v_warehouse_services WHERE storage_end_date IS NULL"
        ).scalar(),
        "customs": db.execute(
            "SELECT COUNT(*) FROM v_customs_services"
        ).scalar(),
        "packing": db.execute(
            "SELECT COUNT(*) FROM v_packing_services"
        ).scalar(),
    }
```

### 5.2 Service Data APIs

```python
# GET /api/trucking
@router.get("/trucking")
async def get_trucking_services(
    date: Optional[date] = None,
    customer_id: Optional[int] = None,
    status: Optional[str] = None,
    db: Session = Depends(get_db)
):
    """Get trucking services from v_trucking_services view"""
    query = "SELECT * FROM v_trucking_services WHERE 1=1"
    params = {}
    
    if date:
        query += " AND scheduled_date = :date"
        params["date"] = date
    if customer_id:
        query += " AND customer_id = :customer_id"
        params["customer_id"] = customer_id
    if status:
        query += " AND status_code = :status"
        params["status"] = status
    
    return db.execute(text(query), params).fetchall()


# Similar endpoints for warehouse, customs, packing...
```
